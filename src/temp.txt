/*
 * ================================================================
 * Monotonic timestamp
 * ================================================================
 */
uint64_t nowNs() {

    return static_cast<uint64_t>(
        std::chrono::duration_cast<
            std::chrono::nanoseconds
        >(
            std::chrono::steady_clock::now()
                .time_since_epoch()
        ).count()
    );
}


/*
 * ================================================================
 * Scheduler conversion
 * ================================================================
 */
SchedulingPolicy getSchedulingPolicy(
    const std::string& scheduler
) {

    if (scheduler == "fcfs") {
        return SchedulingPolicy::FCFS;
    }

    if (scheduler == "sjf") {
        return SchedulingPolicy::SJF;
    }

    if (scheduler == "rr") {
        return SchedulingPolicy::RR;
    }

    if (scheduler == "drr") {
        return SchedulingPolicy::DRR;
    }

    throw std::runtime_error(
        "error: unknown scheduler '" +
        scheduler +
        "'"
    );
}


/*
 * ================================================================
 * Send a complete line for GET
 * ================================================================
 *
 * A GET scheduling unit is a complete line.
 *
 * The newline is part of the byte count.
 */
bool sendGetLine(
    int fd,
    const std::string& line
) {

    if (line.empty()) {
        return true;
    }

    return sendAll(
        fd,
        line.data(),
        line.size()
    );
}


/*
 * ================================================================
 * Prepare GET response
 * ================================================================
 */
bool startGetResponse(
    Request& request,
    const ServerOptions& args
) {

    if (request.response_started) {
        return true;
    }


    const std::string header =
        "OK " +
        std::to_string(request.bytes) +
        "\n";


    if (
        !sendAll(
            request.client_fd,
            header.data(),
            header.size()
        )
    ) {

        return false;
    }


    request.response_started = true;

    return true;
}


/*
 * ================================================================
 * Read next complete GET line
 * ================================================================
 *
 * Uses request.file_offset so a request can be resumed after
 * preemption.
 *
 * The returned line includes '\n' if one exists.
 *
 * The final non-newline-terminated line is also returned.
 */
bool readNextGetLine(
    Request& request,
    std::ifstream& input,
    std::string& line
) {

    /*
     * If a previous round read a line but could not send it,
     * use that line first.
     */
    if (!request.pending_line.empty()) {

        line =
            request.pending_line;

        request.pending_line.clear();

        return true;
    }


    /*
     * Seek to the saved position.
     */
    input.clear();

    input.seekg(
        static_cast<std::streamoff>(
            request.file_offset
        ),
        std::ios::beg
    );


    if (!input) {
        return false;
    }


    line.clear();


    /*
     * Read one character at a time so that the exact
     * line boundary and byte count are known.
     */
    char ch;

    while (input.get(ch)) {

        line.push_back(ch);

        request.file_offset++;


        /*
         * Complete line.
         */
        if (ch == '\n') {
            return true;
        }
    }


    /*
     * Final non-newline line.
     */
    if (!line.empty()) {
        return true;
    }


    return false;
}


/*
 * ================================================================
 * Serve one GET RR round
 * ================================================================
 *
 * Returns:
 *
 *     true  -> request completed
 *     false -> request must be requeued OR failed
 *
 * failed is used to distinguish completion from a socket/file
 * failure.
 */
bool serveGetRR(
    Request& request,
    const ServerOptions& args,
    bool& failed
) {

    failed = false;


    if (
        !startGetResponse(
            request,
            args
        )
    ) {

        failed = true;
        return false;
    }


    const std::size_t quantum =
        args.quantum;


    std::size_t allowance =
        quantum;


    request.rounds++;


    const std::string path =
        args.file_dir +
        "/" +
        request.filename;


    std::ifstream input(
        path,
        std::ios::binary
    );


    if (!input.is_open()) {

        failed = true;
        return false;
    }


    while (
        request.bytes_served <
        request.bytes
    ) {

        std::string line;


        if (
            !readNextGetLine(
                request,
                input,
                line
            )
        ) {

            /*
             * File ended before the declared size.
             */
            input.close();

            failed = true;
            return false;
        }


        const std::size_t line_size =
            line.size();


        /*
         * --------------------------------------------------------
         * Entire request is already complete.
         * --------------------------------------------------------
         */
        if (line_size == 0) {
            continue;
        }


        /*
         * --------------------------------------------------------
         * A14:
         *
         * If the line itself is larger than the entire quantum,
         * RR must send the whole line and end the round.
         * --------------------------------------------------------
         */
        if (line_size > quantum) {

            if (
                !sendGetLine(
                    request.client_fd,
                    line
                )
            ) {

                input.close();

                failed = true;
                return false;
            }


            request.bytes_served +=
                line_size;


            /*
             * This is the amount by which this line exceeded Q.
             *
             * We record it as forfeited/overrun accounting.
             */
            request.forfeited_bytes +=
                line_size - quantum;


            std::cout
                << "[RR A14] Request "
                << request.id
                << " line="
                << line_size
                << " quantum="
                << quantum
                << " overrun="
                << (line_size - quantum)
                << std::endl;


            /*
             * Round ends immediately.
             */
            break;
        }


        /*
         * --------------------------------------------------------
         * Line fits in a fresh quantum but not in the remaining
         * allowance.
         *
         * Do not send a partial line.
         *
         * Save the complete line for the next round.
         * --------------------------------------------------------
         */
        if (line_size > allowance) {

            request.pending_line =
                line;


            /*
             * Remaining allowance is forfeited.
             */
            request.forfeited_bytes +=
                allowance;


            break;
        }


        /*
         * --------------------------------------------------------
         * Normal line transmission.
         * --------------------------------------------------------
         */
        if (
            !sendGetLine(
                request.client_fd,
                line
            )
        ) {

            input.close();

            failed = true;
            return false;
        }


        request.bytes_served +=
            line_size;


        allowance -=
            line_size;


        /*
         * No remaining quantum.
         */
        if (allowance == 0) {
            break;
        }
    }


    input.close();


    /*
     * Request finished.
     */
    return (
        request.bytes_served >=
        request.bytes
    );
}


/*
 * ================================================================
 * Serve one GET DRR round
 * ================================================================
 *
 * DRR retains unused allowance as deficit.
 *
 * Unlike RR, a line larger than the current allowance is NOT
 * forcibly sent. The request accumulates deficit over rounds.
 */
bool serveGetDRR(
    Request& request,
    const ServerOptions& args,
    bool& failed
) {

    failed = false;


    if (
        !startGetResponse(
            request,
            args
        )
    ) {

        failed = true;
        return false;
    }


    const std::size_t quantum =
        args.quantum;


    /*
     * Add quantum to this request's deficit.
     */
    request.deficit +=
        quantum;


    request.rounds++;


    const std::string path =
        args.file_dir +
        "/" +
        request.filename;


    std::ifstream input(
        path,
        std::ios::binary
    );


    if (!input.is_open()) {

        failed = true;
        return false;
    }


    while (
        request.bytes_served <
        request.bytes
    ) {

        std::string line;


        if (
            !readNextGetLine(
                request,
                input,
                line
            )
        ) {

            input.close();

            failed = true;
            return false;
        }


        const std::size_t line_size =
            line.size();


        if (line_size == 0) {
            continue;
        }


        /*
         * --------------------------------------------------------
         * DRR:
         *
         * A line must fit completely inside the current deficit.
         *
         * If it doesn't fit, retain the deficit and let the
         * request accumulate more on a future round.
         * --------------------------------------------------------
         */
        if (line_size > request.deficit) {

            request.pending_line =
                line;

            break;
        }


        /*
         * Send the complete line.
         */
        if (
            !sendGetLine(
                request.client_fd,
                line
            )
        ) {

            input.close();

            failed = true;
            return false;
        }


        request.bytes_served +=
            line_size;


        request.deficit -=
            line_size;


        /*
         * Continue while another complete line fits.
         */
    }


    input.close();


    return (
        request.bytes_served >=
        request.bytes
    );
}


/*
 * ================================================================
 * Serve PUT RR round
 * ================================================================
 *
 * PUT is byte-based, not line-based.
 */
bool servePutRR(
    Request& request,
    const ServerOptions& args,
    bool& failed
) {

    (void)args;

    failed = false;


    /*
     * Send the initial PUT acknowledgement only once.
     */
    if (!request.put_ready_sent) {

        const std::string response =
            "OK 0\n";


        if (
            !sendAll(
                request.client_fd,
                response.data(),
                response.size()
            )
        ) {

            failed = true;
            return false;
        }


        request.put_ready_sent = true;
    }


    request.rounds++;


    const std::size_t remaining =
        request.bytes -
        request.bytes_served;


    const std::size_t amount =
        std::min(
            remaining,
            args.quantum
        );


    if (amount == 0) {

        return true;
    }


    std::vector<char> buffer(
        std::min<std::size_t>(
            64 * 1024,
            amount
        )
    );


    std::size_t left =
        amount;


    while (left > 0) {

        const std::size_t chunk =
            std::min(
                left,
                buffer.size()
            );


        if (
            !recvAll(
                request.client_fd,
                buffer.data(),
                chunk
            )
        ) {

            failed = true;
            return false;
        }


        left -= chunk;


        request.bytes_served +=
            chunk;
    }


    /*
     * PUT request finished.
     */
    if (
        request.bytes_served >=
        request.bytes
    ) {

        const std::string final_response =
            "OK 0\n";


        if (
            !sendAll(
                request.client_fd,
                final_response.data(),
                final_response.size()
            )
        ) {

            failed = true;
            return false;
        }


        return true;
    }


    /*
     * Not finished.
     *
     * Remaining body bytes stay in the TCP receive buffer or
     * continue arriving while the request waits for its next turn.
     */
    return false;
}


/*
 * ================================================================
 * Serve PUT DRR round
 * ================================================================
 */
bool servePutDRR(
    Request& request,
    const ServerOptions& args,
    bool& failed
) {

    (void)args;

    failed = false;


    if (!request.put_ready_sent) {

        const std::string response =
            "OK 0\n";


        if (
            !sendAll(
                request.client_fd,
                response.data(),
                response.size()
            )
        ) {

            failed = true;
            return false;
        }


        request.put_ready_sent = true;
    }


    /*
     * Add quantum to deficit.
     */
    request.deficit +=
        args.quantum;


    request.rounds++;


    const std::size_t remaining =
        request.bytes -
        request.bytes_served;


    const std::size_t amount =
        std::min(
            remaining,
            request.deficit
        );


    if (amount == 0) {
        return false;
    }


    std::vector<char> buffer(
        std::min<std::size_t>(
            64 * 1024,
            amount
        )
    );


    std::size_t left =
        amount;


    while (left > 0) {

        const std::size_t chunk =
            std::min(
                left,
                buffer.size()
            );


        if (
            !recvAll(
                request.client_fd,
                buffer.data(),
                chunk
            )
        ) {

            failed = true;
            return false;
        }


        left -= chunk;


        request.bytes_served +=
            chunk;


        request.deficit -=
            chunk;
    }


    if (
        request.bytes_served >=
        request.bytes
    ) {

        const std::string final_response =
            "OK 0\n";


        if (
            !sendAll(
                request.client_fd,
                final_response.data(),
                final_response.size()
            )
        ) {

            failed = true;
            return false;
        }


        return true;
    }


    return false;
}


/*
 * ================================================================
 * Process one scheduled request
 * ================================================================
 *
 * Returns:
 *
 *     true  -> request completed
 *     false -> request remains unfinished or failed
 *
 * finished / failed are reported separately.
 */
bool processRequest(
    const RequestPtr& request,
    const ServerOptions& args,
    bool& failed
) {

    failed = false;


    if (!request) {
        return true;
    }


    /*
     * ------------------------------------------------------------
     * FCFS / SJF
     * ------------------------------------------------------------
     *
     * For now these continue to serve the whole request.
     */
    if (
        args.sched == "fcfs" ||
        args.sched == "sjf"
    ) {

        /*
         * PUT
         */
        if (request->type == RequestType::PUT) {

            if (!request->put_ready_sent) {

                const std::string response =
                    "OK 0\n";


                if (
                    !sendAll(
                        request->client_fd,
                        response.data(),
                        response.size()
                    )
                ) {

                    failed = true;
                    return false;
                }


                request->put_ready_sent = true;
            }


            const std::string path =
                args.file_dir +
                "/" +
                request->filename;


            std::ofstream output(
                path,
                std::ios::binary
            );


            if (!output.is_open()) {

                failed = true;
                return false;
            }


            constexpr std::size_t BUFFER_SIZE =
                64 * 1024;


            std::vector<char> buffer(
                BUFFER_SIZE
            );


            std::size_t remaining =
                request->bytes -
                request->bytes_served;


            while (remaining > 0) {

                const std::size_t chunk =
                    std::min(
                        BUFFER_SIZE,
                        remaining
                    );


                if (
                    !recvAll(
                        request->client_fd,
                        buffer.data(),
                        chunk
                    )
                ) {

                    output.close();

                    failed = true;
                    return false;
                }


                output.write(
                    buffer.data(),
                    static_cast<std::streamsize>(
                        chunk
                    )
                );


                if (!output) {

                    output.close();

                    failed = true;
                    return false;
                }


                request->bytes_served +=
                    chunk;


                remaining -=
                    chunk;
            }


            output.close();


            const std::string final_response =
                "OK 0\n";


            if (
                !sendAll(
                    request->client_fd,
                    final_response.data(),
                    final_response.size()
                )
            ) {

                failed = true;
                return false;
            }


            return true;
        }


        /*
         * GET
         */
        if (request->type == RequestType::GET) {

            const std::string path =
                args.file_dir +
                "/" +
                request->filename;


            std::ifstream input(
                path,
                std::ios::binary
            );


            if (!input.is_open()) {

                failed = true;
                return false;
            }


            const std::string header =
                "OK " +
                std::to_string(
                    request->bytes
                ) +
                "\n";


            if (
                !sendAll(
                    request->client_fd,
                    header.data(),
                    header.size()
                )
            ) {

                input.close();

                failed = true;
                return false;
            }


            constexpr std::size_t BUFFER_SIZE =
                64 * 1024;


            std::vector<char> buffer(
                BUFFER_SIZE
            );


            std::size_t remaining =
                request->bytes;


            while (remaining > 0) {

                const std::size_t chunk =
                    std::min(
                        BUFFER_SIZE,
                        remaining
                    );


                input.read(
                    buffer.data(),
                    static_cast<std::streamsize>(
                        chunk
                    )
                );


                const std::streamsize received =
                    input.gcount();


                if (received <= 0) {

                    input.close();

                    failed = true;
                    return false;
                }


                if (
                    !sendAll(
                        request->client_fd,
                        buffer.data(),
                        static_cast<std::size_t>(
                            received
                        )
                    )
                ) {

                    input.close();

                    failed = true;
                    return false;
                }


                request->bytes_served +=
                    static_cast<std::size_t>(
                        received
                    );


                remaining -=
                    static_cast<std::size_t>(
                        received
                    );
            }


            input.close();

            return true;
        }


        failed = true;

        return false;
    }


    /*
     * ------------------------------------------------------------
     * RR
     * ------------------------------------------------------------
     */
    if (args.sched == "rr") {

        if (request->type == RequestType::GET) {

            return serveGetRR(
                *request,
                args,
                failed
            );
        }


        if (request->type == RequestType::PUT) {

            return servePutRR(
                *request,
                args,
                failed
            );
        }
    }


    /*
     * ------------------------------------------------------------
     * DRR
     * ------------------------------------------------------------
     */
    if (args.sched == "drr") {

        if (request->type == RequestType::GET) {

            return serveGetDRR(
                *request,
                args,
                failed
            );
        }


        if (request->type == RequestType::PUT) {

            return servePutDRR(
                *request,
                args,
                failed
            );
        }
    }


    failed = true;

    return false;
}


/*
 * ================================================================
 * Main
 * ================================================================
 */
int main(
    int argc,
    char* argv[]
) {

    try {

        /*
         * --------------------------------------------------------
         * Parse arguments
         * --------------------------------------------------------
         */
        ServerOptions args =
            parseServerArgs(
                argc,
                argv
            );


        /*
         * --------------------------------------------------------
         * Validate arguments
         * --------------------------------------------------------
         */
        validateServerOptions(
            args
        );


        /*
         * --------------------------------------------------------
         * Load configuration
         * --------------------------------------------------------
         */
        ServerConfig config =
            loadServerConfig(
                args.config_path
            );


        /*
         * --------------------------------------------------------
         * Scheduler
         * --------------------------------------------------------
         */
        SchedulingPolicy policy =
            getSchedulingPolicy(
                args.sched
            );


        /*
         * --------------------------------------------------------
         * Print configuration
         * --------------------------------------------------------
         */
        std::cout
            << "[Configuration Loaded]"
            << std::endl;

        std::cout
            << "IP: "
            << config.ip
            << std::endl;

        std::cout
            << "Port: "
            << config.port
            << std::endl;

        std::cout
            << "Server threads: "
            << config.server_threads
            << std::endl;

        std::cout
            << "Client threads: "
            << config.client_threads
            << std::endl;

        std::cout
            << "Storage: "
            << args.file_dir
            << std::endl;

        std::cout
            << "Scheduler: "
            << args.sched
            << std::endl;

        if (
            args.sched == "rr" ||
            args.sched == "drr"
        ) {

            std::cout
                << "Quantum: "
                << args.quantum
                << std::endl;
        }


        /*
         * --------------------------------------------------------
         * Create socket
         * --------------------------------------------------------
         */
        int server_fd =
            socket(
                AF_INET,
                SOCK_STREAM,
                0
            );


        if (server_fd < 0) {

            throw std::runtime_error(
                "error: socket creation failed"
            );
        }


        /*
         * --------------------------------------------------------
         * SO_REUSEADDR
         * --------------------------------------------------------
         */
        int reuse = 1;


        if (
            setsockopt(
                server_fd,
                SOL_SOCKET,
                SO_REUSEADDR,
                &reuse,
                sizeof(reuse)
            ) < 0
        ) {

            close(server_fd);

            throw std::runtime_error(
                "error: setsockopt(SO_REUSEADDR) failed"
            );
        }


        /*
         * --------------------------------------------------------
         * Address
         * --------------------------------------------------------
         */
        sockaddr_in server_addr{};

        server_addr.sin_family =
            AF_INET;

        server_addr.sin_port =
            htons(
                static_cast<uint16_t>(
                    config.port
                )
            );


        if (
            inet_pton(
                AF_INET,
                config.ip.c_str(),
                &server_addr.sin_addr
            ) <= 0
        ) {

            close(server_fd);

            throw std::runtime_error(
                "error: invalid server IP address"
            );
        }


        /*
         * --------------------------------------------------------
         * Bind
         * --------------------------------------------------------
         */
        if (
            bind(
                server_fd,
                reinterpret_cast<sockaddr*>(
                    &server_addr
                ),
                sizeof(server_addr)
            ) < 0
        ) {

            close(server_fd);

            throw std::runtime_error(
                "error: bind failed"
            );
        }


        /*
         * --------------------------------------------------------
         * Listen
         * --------------------------------------------------------
         */
        if (
            listen(
                server_fd,
                128
            ) < 0
        ) {

            close(server_fd);

            throw std::runtime_error(
                "error: listen failed"
            );
        }


        std::cout
            << "[TCP Server Listening] "
            << config.ip
            << ":"
            << config.port
            << std::endl;


        /*
         * --------------------------------------------------------
         * Shared queue
         * --------------------------------------------------------
         */
        RequestQueue queue(policy);


        /*
         * --------------------------------------------------------
         * Request IDs
         * --------------------------------------------------------
         */
        std::atomic<uint64_t>
            next_request_id{1};


        /*
         * --------------------------------------------------------
         * Worker pool
         * --------------------------------------------------------
         */
        std::vector<std::thread> workers;


        workers.reserve(
            static_cast<std::size_t>(
                config.server_threads
            )
        );


        for (
            int i = 0;
            i < config.server_threads;
            ++i
        ) {

            workers.emplace_back(
                [&queue, &args]() {

                    while (true) {

                        RequestPtr request =
                            queue.pop();


                        if (!request) {
                            break;
                        }


                        std::cout
                            << "[Worker "
                            << std::this_thread::get_id()
                            << "] Request "
                            << request->id
                            << " ";

                        if (
                            request->type ==
                            RequestType::GET
                        ) {

                            std::cout
                                << "GET "
                                << request->filename;
                        }
                        else if (
                            request->type ==
                            RequestType::PUT
                        ) {

                            std::cout
                                << "PUT "
                                << request->filename;
                        }


                        std::cout
                            << " ("
                            << request->bytes
                            << " bytes)"
                            << std::endl;


                        bool failed = false;


                        bool finished =
                            processRequest(
                                request,
                                args,
                                failed
                            );


                        /*
                         * ------------------------------------------------
                         * Failure
                         * ------------------------------------------------
                         */
                        if (failed) {

                            close(
                                request->client_fd
                            );


                            std::cerr
                                << "[Worker] Request "
                                << request->id
                                << " failed"
                                << std::endl;


                            continue;
                        }


                        /*
                         * ------------------------------------------------
                         * Request completed
                         * ------------------------------------------------
                         */
                        if (finished) {

                            close(
                                request->client_fd
                            );


                            std::cout
                                << "[Worker] Completed request "
                                << request->id
                                << std::endl;


                            continue;
                        }


                        /*
                         * ------------------------------------------------
                         * Request was preempted.
                         * ------------------------------------------------
                         *
                         * The connection remains open.
                         *
                         * The request is put back into the shared
                         * scheduler queue.
                         */
                        std::cout
                            << "[Preempted] Request "
                            << request->id
                            << " served="
                            << request->bytes_served
                            << "/"
                            << request->bytes
                            << " bytes"
                            << " round="
                            << request->rounds
                            << std::endl;


                        queue.push(
                            request
                        );
                    }
                }
            );
        }


        /*
         * --------------------------------------------------------
         * Admission loop
         * --------------------------------------------------------
         */
        while (true) {

            sockaddr_in client_addr{};

            socklen_t client_len =
                sizeof(client_addr);


            int client_fd =
                accept(
                    server_fd,
                    reinterpret_cast<sockaddr*>(
                        &client_addr
                    ),
                    &client_len
                );


            if (client_fd < 0) {

                if (errno == EINTR) {
                    continue;
                }


                std::cerr
                    << "error: accept failed"
                    << std::endl;

                break;
            }


            /*
             * ----------------------------------------------------
             * Read header
             * ----------------------------------------------------
             */
            std::string header;


            if (
                !recvLine(
                    client_fd,
                    header
                )
            ) {

                const std::string error =
                    "ERR Invalid request\n";


                sendAll(
                    client_fd,
                    error.data(),
                    error.size()
                );


                close(client_fd);

                continue;
            }


            /*
             * ----------------------------------------------------
             * Parse protocol
             * ----------------------------------------------------
             */
            ParsedRequest parsed;


            try {

                parsed =
                    parseProtocolRequest(
                        header
                    );
            }
            catch (
                const std::exception& e
            ) {

                const std::string error =
                    std::string("ERR ") +
                    e.what() +
                    "\n";


                sendAll(
                    client_fd,
                    error.data(),
                    error.size()
                );


                close(client_fd);

                continue;
            }


            /*
             * ----------------------------------------------------
             * HEALTH
             * ----------------------------------------------------
             */
            if (
                parsed.type ==
                RequestType::HEALTH
            ) {

                const std::string response =
                    "OK " +
                    std::to_string(
                        queue.size()
                    ) +
                    "\n";


                sendAll(
                    client_fd,
                    response.data(),
                    response.size()
                );


                close(client_fd);

                continue;
            }


            /*
             * ----------------------------------------------------
             * Create request
             * ----------------------------------------------------
             */
            auto request =
                std::make_shared<Request>();


            request->id =
                next_request_id.fetch_add(1);


            request->type =
                parsed.type;


            request->filename =
                parsed.filename;


            request->client_fd =
                client_fd;


            request->arrival_ns =
                nowNs();


            /*
             * ----------------------------------------------------
             * GET size
             * ----------------------------------------------------
             */
            if (
                request->type ==
                RequestType::GET
            ) {

                const std::string path =
                    args.file_dir +
                    "/" +
                    request->filename;


                std::ifstream input(
                    path,
                    std::ios::binary |
                    std::ios::ate
                );


                if (!input.is_open()) {

                    const std::string error =
                        "ERR File not found\n";


                    sendAll(
                        client_fd,
                        error.data(),
                        error.size()
                    );


                    close(client_fd);

                    continue;
                }


                const std::streamoff file_size =
                    input.tellg();


                input.close();


                if (file_size < 0) {

                    const std::string error =
                        "ERR Cannot determine file size\n";


                    sendAll(
                        client_fd,
                        error.data(),
                        error.size()
                    );


                    close(client_fd);

                    continue;
                }


                request->bytes =
                    static_cast<std::size_t>(
                        file_size
                    );
            }


            /*
             * ----------------------------------------------------
             * PUT size
             * ----------------------------------------------------
             */
            else if (
                request->type ==
                RequestType::PUT
            ) {

                request->bytes =
                    parsed.bytes;
            }


            /*
             * ----------------------------------------------------
             * Admission log
             * ----------------------------------------------------
             */
            std::cout
                << "[Admitted] Request "
                << request->id
                << " ";


            if (
                request->type ==
                RequestType::GET
            ) {

                std::cout
                    << "GET "
                    << request->filename;
            }
            else if (
                request->type ==
                RequestType::PUT
            ) {

                std::cout
                    << "PUT "
                    << request->filename;
            }


            std::cout
                << " ("
                << request->bytes
                << " bytes)"
                << std::endl;


            /*
             * ----------------------------------------------------
             * Add to scheduler.
             * ----------------------------------------------------
             */
            queue.push(
                request
            );
        }


        /*
         * --------------------------------------------------------
         * Stop accepting.
         * --------------------------------------------------------
         */
        close(server_fd);


        /*
         * --------------------------------------------------------
         * Drain scheduler and stop workers.
         * --------------------------------------------------------
         */
        queue.shutdown();


        for (auto& worker : workers) {

            if (worker.joinable()) {
                worker.join();
            }
        }


        std::cout
            << "[Server Shutdown]"
            << std::endl;


        return 0;
    }


    catch (
        const std::exception& e
    ) {

        std::cerr
            << e.what()
            << std::endl;

        return 1;
    }
}
